I'm using Docker on Mac with the following commands `make build && make simulate`.
The outputs (JSON and HTML) will be available in `.sim_results/` folder

**Makefile**
```
.PHONY: build run shell simulate

build:
	docker build -t pokemon-agent .

simulate:
	mkdir -p .sim_results
	docker run --rm -v "$$(PWD)/.sim_results:/output" pokemon-agent python simulate.py
```

**Dockerfile**
```
# Stage 1: Build the cabt visualizer
FROM --platform=linux/amd64 node:20-slim AS visualizer-builder

RUN apt-get update && apt-get install -y git && rm -rf /var/lib/apt/lists/*

# Clone the kaggle-environments repo (shallow for speed)
RUN git clone --depth 1 https://github.com/Kaggle/kaggle-environments.git /repo

# Install pnpm
RUN corepack enable && corepack prepare pnpm@latest --activate

WORKDIR /repo

# Install workspace dependencies and build the cabt visualizer
RUN pnpm install --frozen-lockfile --ignore-scripts 2>&1 | tail -5
RUN pnpm --filter @kaggle-environments/cabt-visualizer run build 2>&1 | tail -10

# Stage 2: Main application
FROM --platform=linux/amd64 python:3.13-slim

# Install uv
COPY --from=ghcr.io/astral-sh/uv:latest /uv /uvx /bin/

# Copy project files
WORKDIR /app
COPY sample_submission/ /app/
COPY pyproject.toml simulate.py /app/

# Copy the built visualizer into the kaggle-environments package installed later
# We'll do this after pip install

# Install dependencies
RUN uv pip install --system kaggle-environments>=1.30.1

# Copy the built visualizer dist into the installed package
COPY --from=visualizer-builder /repo/kaggle_environments/envs/cabt/visualizer/default/dist/ \
    /usr/local/lib/python3.13/site-packages/kaggle_environments/envs/cabt/visualizer/default/dist/

RUN ls -la /usr/local/lib/python3.13/site-packages/kaggle_environments/envs/cabt/visualizer/default/dist/ 2>&1 || echo "No dist directory"

CMD ["python", "main.py"]
```

**simulate.py**
```
#!/usr/bin/env python3
"""Run a cabt simulation and produce a standalone HTML with the visualizer."""
import json
import os
import re
import sys

from kaggle_environments import make
from main import agent

out_dir = os.environ.get("OUTPUT_DIR", "/output")
os.makedirs(out_dir, exist_ok=True)

with open("deck.csv") as f:
    deck = [int(line) for line in f.readlines() if line.strip()]

print("Starting simulation...", flush=True)
env = make("cabt", configuration={"decks": [deck, deck]})
env.run([agent, agent])

# Read the built visualizer HTML
viz_path = "/usr/local/lib/python3.13/site-packages/kaggle_environments/envs/cabt/visualizer/default/dist/index.html"
if not os.path.exists(viz_path):
    print(f"ERROR: visualizer not found at {viz_path}", flush=True)
    sys.exit(1)

with open(viz_path, encoding="utf-8") as f:
    viz_html = f.read()

# Extract the environment data and agent info from kaggle replay
replay_html = env.render(mode="html")

# Parse it to get the full JSON
start = replay_html.find("window.kaggle = ")
end = replay_html.find(";\n      </script>", start)
if start >= 0 and end >= 0:
    kaggle_json_str = replay_html[start + len("window.kaggle = "):end]
    window_kaggle = json.loads(kaggle_json_str)
else:
    window_kaggle = {
        "debug": False,
        "playing": False,
        "step": len(env.steps) - 1,
        "controls": True,
        "environment": env.toJSON(),
        "logs": env.logs,
    }

# Build the postMessage data payload that the visualizer expects
# The visualizer listens for: agents, environment (with steps), step
env_data = window_kaggle.get("environment", env.toJSON())
agents = window_kaggle.get("agents", [])
step = window_kaggle.get("step", len(env_data.get("steps", [])) - 1)

# Create a postMessage bridge: after the module script loads, simulate
# the parent iframe sending the replay data
bridge_script = f"""
<script>
// Wait for page to load, then send data as postMessage
(function() {{
    var data = {json.dumps({
        "agents": agents,
        "environment": env_data,
        "step": step,
    })};
    setTimeout(function() {{
        window.dispatchEvent(new MessageEvent('message', {{
            data: data,
            origin: window.location.origin,
            source: window,
        }}));
    }}, 100);
}})();
</script>
"""

# Inject bridge script before closing </body>
viz_html = viz_html.replace("</body>", bridge_script + "\n</body>")

out_path = os.path.join(out_dir, "result.html")
with open(out_path, "w", encoding="utf-8") as f:
    f.write(viz_html)

# Also export the raw replay JSON
json_path = os.path.join(out_dir, "replay.json")
with open(json_path, "w", encoding="utf-8") as f:
    json.dump(window_kaggle, f, indent=2)
print(f"Done!")
print(f"  HTML visualizer: {out_path}")
print(f"  JSON replay:     {json_path}")
```
