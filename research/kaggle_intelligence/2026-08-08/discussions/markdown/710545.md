This is a Python code that I use myself, which can be used to translate the replay into readable text, and then you can give the text to AI to let it translate for you.(You should rename the replay "log.json".my English is not so good)

`import json

def describe(log):
    t = log.get('type')
    p = log.get('playerIndex')
    if t == 10:
        return f"Player {p} drew a card."
    if t == 8:
        return f"Player {p} switched active Pokemon."
    if t == 15:
        return f"Player {p} attacked (Card ID {log.get('cardId')})."
    if t == 16:
        return f"Player {p}'s Pokemon took {abs(log.get('value', 0))} damage."
    if t == 6:
        c, f, to = log.get('cardId'), log.get('fromArea'), log.get('toArea')
        if f == 1 and to == 2:
            return f"Player {p} played card {c}."
        if f == 4 and to == 3:
            return f"Player {p}'s card {c} was knocked out."
        return f"Player {p} moved card {c} (area {f}->{to})."
    return None

# 读取文件
with open('log.json', 'r', encoding='utf-8') as f:
    data = json.load(f)

# ----- 递归遍历所有字典和列表，收集所有 'logs' 列表 -----
all_logs = []

def collect_logs(obj):
    if isinstance(obj, dict):
        for key, value in obj.items():
            if key == 'logs' and isinstance(value, list):
                all_logs.extend(value)
            else:
                collect_logs(value)
    elif isinstance(obj, list):
        for item in obj:
            collect_logs(item)

collect_logs(data)
print(f"✅ 共收集到 {len(all_logs)} 条日志")

# 过滤并输出
filtered = []
for log in all_logs:
    desc = describe(log)
    if desc:
        filtered.append(desc)

print(f"✅ 过滤后保留 {len(filtered)} 条事件\n")
for desc in filtered:
    print(desc)`
