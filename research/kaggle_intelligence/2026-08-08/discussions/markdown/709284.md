Thank you! If it helps others, I've since produced a script to enter a pokemon ID and receive a printout of the pokemon's attacks. *Apologies, it's not formatted within the code block correctly.*

`
from cg import api
def find_card(all_cards, selected_card):
    selected_card = selected_card.strip().lower()

    for card in all_cards:
        if str(card.cardId).lower() == selected_card:
            # Return the selected card object once match is found
            return card 
    return None
def main():

    # Gets all attacks and cards from the API and creates a dictionary of attacks by attackId for easy lookup.
    all_attack_ids = {a.attackId: a for a in api.all_attack()}
    all_cards = api.all_card_data()

    # Prompt user to select a card by ID. ID can be found in CSV.
    selected_card_id = input("Enter card id: ")
    card = find_card(all_cards, selected_card_id)

    # If card is not found, print message and exit.
    if card is None:
        print("Card not found.")
        return

    # Print card name.
    print(f"Selected card: {card.name}\n")

    # If card has no attacks, print message and exit.
    # This would happen in cases where Energy cards or Trainer cards
    # are selected.
    if not card.attacks:
        print("This card has no attacks.")
        return

    # Print attack IDs and names for the selected card.
    print("Attack Details:")
    for attack_id in card.attacks:
        attack = all_attack_ids.get(attack_id)
        # Print attack attributes: attackId, name, damage, text
        print(
            f"Attack ID: {attack.attackId}\nAttack Name: {attack.name}\n"
            f"Attack Damage: {attack.damage}\nAttack Text: {attack.text}\n"
            f"Attack Cost: {attack.energies}"
        )
if __name__ == "__main__":
    main()
`
