vibe coded (kimi 2.7) limitless to number.csv
```python
#!/usr/bin/env python3
"""Convert a Limitless TCG deck list into the Kaggle PTCG 60-line deck.csv format.

Usage:
    uv run python convert_deck.py
    uv run python convert_deck.py --deck deck.txt --cards EN_Card_Data.csv --output deck.csv
"""

import argparse
import csv
import difflib
import re
import sys
from pathlib import Path

# Limitless "<Type> Energy" names map to the CSV's "Basic {<symbol>} Energy" names.
BASIC_ENERGY_MAP = {
    "Grass Energy": "Basic {G} Energy",
    "Fire Energy": "Basic {R} Energy",
    "Water Energy": "Basic {W} Energy",
    "Lightning Energy": "Basic {L} Energy",
    "Psychic Energy": "Basic {P} Energy",
    "Fighting Energy": "Basic {F} Energy",
    "Darkness Energy": "Basic {D} Energy",
    "Metal Energy": "Basic {M} Energy",
}


def normalize_name(name: str) -> str:
    """Normalize card names so that curly apostrophes match straight ones."""
    return (
        name.strip()
        .replace("’", "'")  # U+2019 right single quotation mark
        .replace("‘", "'")  # U+2018 left single quotation mark
        .replace("“", '"')
        .replace("”", '"')
    )


def load_card_database(csv_path: Path) -> dict:
    """Load EN_Card_Data.csv and build multi-level lookup dictionaries.

    Returns:
        A dict with keys:
            - exact: {(name, expansion, number): card_id}
            - name_expansion: {(name, expansion): card_id}
            - name: {name: card_id}
            - all_names: list of normalized card names for fuzzy matching
    """
    exact: dict[tuple[str, str, str], str] = {}
    by_name_expansion: dict[tuple[str, str], str] = {}
    by_name: dict[str, str] = {}
    all_names: set[str] = set()

    with csv_path.open(encoding="utf-8", newline="") as file:
        reader = csv.DictReader(file)
        for row in reader:
            name = normalize_name(row["Card Name"])
            expansion = row["Expansion"].strip()
            number = row["Collection No."].strip()
            card_id = row["Card ID"].strip()

            all_names.add(name)

            exact_key = (name, expansion, number)
            if exact_key not in exact:
                exact[exact_key] = card_id

            ne_key = (name, expansion)
            if ne_key not in by_name_expansion:
                by_name_expansion[ne_key] = card_id

            if name not in by_name:
                by_name[name] = card_id

    return {
        "exact": exact,
        "name_expansion": by_name_expansion,
        "name": by_name,
        "all_names": sorted(all_names),
    }


def parse_deck_line(line: str) -> tuple[int, str, str, str]:
    """Parse one deck list line into (quantity, name, expansion, number).

    Limitless format: ``<quantity> <card name> <set> <number>``.
    The card name may contain spaces, so we split from the right for the
    expansion code and collection number.
    """
    line = line.strip()
    if not line:
        raise ValueError("Empty line")

    parts = line.split(" ", 1)
    if len(parts) != 2:
        raise ValueError(f"Cannot parse line: {line!r}")

    quantity_str, rest = parts
    quantity = int(quantity_str)

    # Name can contain spaces; set and number are the last two tokens.
    try:
        name, expansion, number = rest.rsplit(" ", 2)
    except ValueError as exc:
        raise ValueError(f"Cannot parse card/set/number from: {line!r}") from exc

    return quantity, normalize_name(name), expansion.strip(), number.strip()


def parse_deck(deck_path: Path) -> list[tuple[int, str, str, str]]:
    """Parse a Limitless deck file into a list of (quantity, name, expansion, number)."""
    entries: list[tuple[int, str, str, str]] = []
    section_header = re.compile(r"^(Pokémon|Pokemon|Trainer|Energy):\s*\d+", re.IGNORECASE)

    for raw in deck_path.read_text(encoding="utf-8").splitlines():
        line = raw.strip()
        if not line or section_header.match(line):
            continue
        entries.append(parse_deck_line(line))

    return entries


def resolve_card(
    name: str,
    expansion: str,
    number: str,
    db: dict,
) -> tuple[str | None, str]:
    """Resolve a single card to its Card ID.

    Returns:
        (card_id, match_type). ``card_id`` is None if no match was found.
    """
    exact = db["exact"]
    by_name_expansion = db["name_expansion"]
    by_name = db["name"]

    key = (name, expansion, number)
    if key in exact:
        return exact[key], "exact"

    ne_key = (name, expansion)
    if ne_key in by_name_expansion:
        return by_name_expansion[ne_key], "name+expansion"

    if name in by_name:
        return by_name[name], "name-only"

    # Basic energy names in Limitless differ from the CSV names.
    energy_name = BASIC_ENERGY_MAP.get(name)
    if energy_name and energy_name in by_name:
        return by_name[energy_name], "energy-name-mapping"

    # Fuzzy name match as a last resort (e.g. "Telepathic" vs "Telepath").
    close = difflib.get_close_matches(name, db["all_names"], n=1, cutoff=0.85)
    if close:
        return by_name[close[0]], f"fuzzy-name ({close[0]})"

    return None, "unmatched"


def convert_deck(deck_path: Path, cards_path: Path, output_path: Path) -> None:
    """Convert deck.txt to deck.csv using EN_Card_Data.csv."""
    db = load_card_database(cards_path)
    entries = parse_deck(deck_path)

    card_ids: list[str] = []
    unmatched: list[tuple[str, int, str, str, str]] = []  # line, qty, name, exp, no
    fallbacks: list[tuple[str, int, str, str, str, str, str]] = []  # line, qty, name, exp, no, matched_id, strategy

    for quantity, name, expansion, number in entries:
        card_id, strategy = resolve_card(name, expansion, number, db)
        line_text = f"{quantity} {name} {expansion} {number}"

        if card_id is None:
            unmatched.append((line_text, quantity, name, expansion, number))
            continue

        if strategy != "exact":
            fallbacks.append((line_text, quantity, name, expansion, number, card_id, strategy))

        card_ids.extend([card_id] * quantity)

    # Report issues.
    if fallbacks:
        print("Cards matched via fallback rules:", file=sys.stderr)
        for line, qty, name, exp, no, cid, strategy in fallbacks:
            print(f"  [{strategy:17}] {line} -> ID {cid}", file=sys.stderr)

    if unmatched:
        print("\nUnmatched cards (no Card ID found):", file=sys.stderr)
        for line, qty, name, exp, no in unmatched:
            print(f"  {line}", file=sys.stderr)
        raise SystemExit(f"\nError: {len(unmatched)} card(s) could not be matched.")

    if len(card_ids) != 60:
        raise SystemExit(f"Error: deck contains {len(card_ids)} cards, expected exactly 60.")

    output_path.write_text("\n".join(card_ids) + "\n", encoding="utf-8")
    print(f"Wrote {len(card_ids)} card IDs to {output_path}")


def main() -> None:
    parser = argparse.ArgumentParser(description="Convert a Limitless TCG deck list to Kaggle PTCG deck.csv format.")
    parser.add_argument("--deck", type=Path, default=Path("deck.txt"))
    parser.add_argument("--cards", type=Path, default=Path("EN_Card_Data.csv"))
    parser.add_argument("--output", type=Path, default=Path("deck.csv"))
    args = parser.parse_args()

    convert_deck(args.deck, args.cards, args.output)


if __name__ == "__main__":
    main()

```
